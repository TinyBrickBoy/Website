{{- /* Sammelt die Statistik einer Plattform. Aufruf: partial "stats/collect.html" (dict "kind" "github"|"codeberg" "conf" site.Params.stats.github) Reihenfolge: data/stats/.json (von scripts/update-data.py) → live von der API → Werte aus hugo.toml. */ -}} {{- $conf := .conf -}} {{- $user := $conf.user -}} {{- $api := cond (eq .kind "github") "https://api.github.com" "https://codeberg.org/api/v1" -}} {{- $s := dict "user" $user "repos" $conf.repos "forks" $conf.forks "stars" $conf.stars "followers" $conf.followers "since" $conf.since "langs" $conf.langs "live" false -}} {{- $file := index (site.Data.stats | default dict) .kind -}} {{- $profile := false -}} {{- $repos := slice -}} {{- if $file -}} {{- $s = merge $s (dict "repos" $file.repos "forks" $file.forks "stars" $file.stars "followers" $file.followers "since" $file.since "langs" $file.langs "fetched" $file.fetched) -}} {{- else -}} {{- $profile = partial "stats/fetch.html" (printf "%s/users/%s" $api $user) -}} {{- $ok := not $file -}} {{- range seq 5 -}} {{- if $ok -}} {{- $url := cond (eq $.kind "github") (printf "%s/users/%s/repos?per_page=100&page=%d" $api $user .) (printf "%s/users/%s/repos?limit=50&page=%d" $api $user .) -}} {{- $page := partial "stats/fetch.html" $url -}} {{- if and $page (gt (len $page) 0) }}{{ $repos = $repos | append $page }}{{ else }}{{ $ok = false }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- if $profile -}} {{- $starsKey := cond (eq .kind "github") "stargazers_count" "stars_count" -}} {{- $followers := cond (eq .kind "github") (index $profile "followers") (index $profile "followers_count") -}} {{- $created := cond (eq .kind "github") (index $profile "created_at") (index $profile "created") -}} {{- $own := where $repos "fork" false -}} {{- $stars := 0 }}{{ range $own }}{{ $stars = add $stars (index . $starsKey) }}{{ end -}} {{- $langCount := dict -}} {{- range $own }}{{ with .language }}{{ $langCount = merge $langCount (dict . (add (index $langCount . | default 0) 1)) }}{{ end }}{{ end -}} {{- $langs := slice -}} {{- range $k, $v := $langCount }}{{ $langs = $langs | append (dict "name" $k "n" $v) }}{{ end -}} {{- $langList := slice -}} {{- range first 6 (sort $langs "n" "desc") }}{{ $langList = $langList | append (slice (slice .name .n)) }}{{ end -}} {{- $s = merge $s (dict "repos" (len $own) "forks" (sub (len $repos) (len $own)) "stars" $stars "followers" $followers "since" (substr $created 0 4) "langs" $langList "live" true) -}} {{- end -}} {{- return $s -}}
